## Usage Example

```c
#include <openssl/ui.h>
#include <openssl/err.h>
#include "uierr.h"

/* Load UI error strings at application startup */
ERR_load_UI_strings();

/* After a failed UI operation, retrieve the human-readable error */
UI *ui = UI_new();
if (UI_process(ui) < 0) {
    unsigned long err = ERR_get_error();
    /* Reports function + reason using registered UI strings */
    fprintf(stderr, "UI error: %s\n", ERR_error_string(err, NULL));
}
UI_free(ui);
```
