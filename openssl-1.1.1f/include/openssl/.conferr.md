## Usage Example

```c
#include <openssl/conf.h>
#include <openssl/conferr.h>
#include <openssl/err.h>

void load_config(const char *path) {
    CONF *conf = NCONF_new(NULL);

    /* Load the configuration file */
    if (!NCONF_load(conf, path, NULL)) {
        /* Retrieve and print the CONF error */
        unsigned long err = ERR_get_error();
        int reason = ERR_GET_REASON(err);

        if (reason == CONF_R_MISSING_EQUAL_SIGN) {
            fprintf(stderr, "Config syntax error: missing '=' sign\n");
        } else if (reason == CONF_R_NO_SUCH_FILE) {
            fprintf(stderr, "Config file not found: %s\n", path);
        } else {
            ERR_print_errors_fp(stderr);
        }
    }

    NCONF_free(conf);
}
```
