## Usage Example

```c
#include <openssl/pem.h>
#include <openssl/err.h>
#include "pemerr.h"

FILE *fp = fopen("key.pem", "r");
EVP_PKEY *pkey = PEM_read_PrivateKey(fp, NULL, NULL, NULL);

if (!pkey) {
    unsigned long err_code = ERR_get_error();
    int reason = ERR_GET_REASON(err_code);

    if (reason == PEM_R_BAD_PASSWORD_READ) {
        fprintf(stderr, "Password prompt failed\n");
    } else if (reason == PEM_R_NO_START_LINE) {
        fprintf(stderr, "File does not contain a PEM-encoded key\n");
    } else {
        ERR_print_errors_fp(stderr);
    }
}

fclose(fp);
```
