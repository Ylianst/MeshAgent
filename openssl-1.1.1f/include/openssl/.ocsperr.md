## Usage Example

```c
#include <openssl/ocsp.h>
#include <openssl/err.h>
#include "ocsperr.h"

/* Load OCSP error strings at application startup */
ERR_load_OCSP_strings();

/* After a failed OCSP operation, retrieve the error */
unsigned long err = ERR_get_error();
int reason = ERR_GET_REASON(err);

if (reason == OCSP_R_CERTIFICATE_VERIFY_ERROR) {
    fprintf(stderr, "OCSP certificate verification failed\n");
} else if (reason == OCSP_R_STATUS_EXPIRED) {
    fprintf(stderr, "OCSP status has expired\n");
}
```
