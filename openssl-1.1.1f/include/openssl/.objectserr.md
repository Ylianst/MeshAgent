## Usage Example

```c
#include <openssl/objects.h>
#include <openssl/objectserr.h>

/* Load OBJ error strings once at startup */
ERR_load_OBJ_strings();

ASN1_OBJECT *obj = OBJ_nid2obj(NID_sha256);
if (obj == NULL) {
    /* ERR_get_error() will now return OBJ_F_OBJ_NID2OBJ
       paired with OBJ_R_UNKNOWN_NID if the NID is invalid */
    unsigned long err = ERR_get_error();
    fprintf(stderr, "OBJ error: %s\n", ERR_error_string(err, NULL));
}
```
