## Usage Example

```c
#include <openssl/conf.h>

/* Load a config file using the modern NCONF API */
CONF *conf = NCONF_new_ex(NULL, NCONF_default());
long eline = 0;

if (!NCONF_load(conf, "/etc/ssl/openssl.cnf", &eline)) {
    fprintf(stderr, "Error on line %ld\n", eline);
    NCONF_free(conf);
    return 1;
}

/* Read a value from [default_sect] */
const char *val = NCONF_get_string(conf, "default_sect", "my_key");
if (val)
    printf("my_key = %s\n", val);

/* Load modules defined in the config */
CONF_modules_load(conf, "myapp", CONF_MFLAGS_IGNORE_MISSING_FILE);

NCONF_free(conf);
```
