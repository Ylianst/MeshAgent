## Usage Example

```c
#include <openssl/rsa.h>
#include <openssl/err.h>
#include <openssl/rsaerr.h>

void check_rsa_error(void) {
    unsigned long err = ERR_get_error();
    int reason = ERR_GET_REASON(err);

    if (reason == RSA_R_KEY_SIZE_TOO_SMALL) {
        fprintf(stderr, "RSA key is too small for this operation\n");
    } else if (reason == RSA_R_PADDING_CHECK_FAILED) {
        fprintf(stderr, "RSA padding verification failed\n");
    } else if (reason == RSA_R_INVALID_PSS_PARAMETERS) {
        fprintf(stderr, "Invalid PSS salt/digest parameters\n");
    }
}
```
