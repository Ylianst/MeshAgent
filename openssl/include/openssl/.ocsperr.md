## Usage Example

```c
#include <openssl/ocsp.h>
#include <openssl/ocsperr.h>
#include <openssl/err.h>

int check_ocsp_error(void) {
    unsigned long err_code = ERR_get_error();
    int reason = ERR_GET_REASON(err_code);

    switch (reason) {
        case OCSP_R_CERTIFICATE_VERIFY_ERROR:
            fprintf(stderr, "OCSP: Certificate verification failed\n");
            break;
        case OCSP_R_SIGNATURE_FAILURE:
            fprintf(stderr, "OCSP: Invalid response signature\n");
            break;
        case OCSP_R_STATUS_EXPIRED:
            fprintf(stderr, "OCSP: Status information has expired\n");
            break;
        case OCSP_R_ROOT_CA_NOT_TRUSTED:
            fprintf(stderr, "OCSP: Root CA is not trusted\n");
            break;
        default:
            fprintf(stderr, "OCSP: Unknown error %d\n", reason);
    }
    return -1;
}
```
