## Usage Example

These codes appear in OpenSSL error stacks when ESS operations fail. Check them using standard OpenSSL error APIs:

```c
#include <openssl/ess.h>
#include <openssl/esserr.h>
#include <openssl/err.h>

int verify_signing_cert(CMS_ContentInfo *cms) {
    if (!CMS_verify(cms, certs, store, NULL, NULL, 0)) {
        unsigned long err = ERR_get_error();
        if (ERR_GET_REASON(err) == ESS_R_ESS_CERT_ID_NOT_FOUND) {
            fprintf(stderr, "Signing certificate not found in ESS attribute\n");
        } else if (ERR_GET_REASON(err) == ESS_R_MISSING_SIGNING_CERTIFICATE_ATTRIBUTE) {
            fprintf(stderr, "SigningCertificate attribute missing\n");
        }
        return 0;
    }
    return 1;
}
```
