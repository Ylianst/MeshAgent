## Usage Example

```c
#include <openssl/x509v3.h>
#include <openssl/err.h>

void check_extension_error(void) {
    unsigned long err = ERR_get_error();

    if (ERR_GET_REASON(err) == X509V3_R_EXTENSION_NOT_FOUND) {
        fprintf(stderr, "Required X.509v3 extension is missing\n");
    }

    if (ERR_GET_REASON(err) == X509V3_R_INVALID_IPADDRESS) {
        fprintf(stderr, "Malformed IP address in certificate extension\n");
    }

    /* Print full human-readable error string */
    ERR_print_errors_fp(stderr);
}
```
