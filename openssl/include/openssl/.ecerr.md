## Usage Example

```c
#include <openssl/ec.h>
#include <openssl/ecerr.h>
#include <openssl/err.h>

void check_ec_error(void) {
    unsigned long err = ERR_get_error();
    int reason = ERR_GET_REASON(err);

    switch (reason) {
        case EC_R_POINT_IS_NOT_ON_CURVE:
            fprintf(stderr, "EC point validation failed: not on curve\n");
            break;
        case EC_R_INVALID_KEY:
            fprintf(stderr, "EC key is invalid\n");
            break;
        case EC_R_CURVE_DOES_NOT_SUPPORT_ECDH:
            fprintf(stderr, "Selected curve cannot be used for ECDH\n");
            break;
        default:
            fprintf(stderr, "Unknown EC error: %lu\n", err);
    }
}
```
