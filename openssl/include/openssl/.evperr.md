```c
#include <openssl/evp.h>
#include <openssl/err.h>
#include <openssl/evperr.h>

EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();
if (EVP_EncryptInit_ex(ctx, EVP_aes_256_cbc(), NULL, key, iv) != 1) {
    unsigned long err_code = ERR_get_error();

    /* Check the reason code against EVP_R_* constants */
    int reason = ERR_GET_REASON(err_code);
    if (reason == EVP_R_INVALID_KEY_LENGTH) {
        fprintf(stderr, "Key length mismatch for selected cipher\n");
    } else if (reason == EVP_R_AES_KEY_SETUP_FAILED) {
        fprintf(stderr, "AES key schedule initialization failed\n");
    }

    ERR_print_errors_fp(stderr);
}
```
