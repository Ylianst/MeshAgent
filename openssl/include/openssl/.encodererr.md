## Usage Example

These codes appear in OpenSSL error diagnostics when encoding operations fail:

```c
#include <openssl/encoder.h>
#include <openssl/encodererr.h>
#include <openssl/err.h>

OSSL_ENCODER_CTX *ctx = OSSL_ENCODER_CTX_new_for_pkey(pkey, OSSL_KEYMGMT_SELECT_ALL,
                                                        "PEM", NULL, NULL);
if (ctx == NULL) {
    unsigned long err = ERR_get_error();

    if (ERR_GET_REASON(err) == OSSL_ENCODER_R_ENCODER_NOT_FOUND) {
        fprintf(stderr, "No encoder found for the requested format\n");
    } else if (ERR_GET_REASON(err) == OSSL_ENCODER_R_INCORRECT_PROPERTY_QUERY) {
        fprintf(stderr, "Invalid property query supplied\n");
    } else if (ERR_GET_REASON(err) == OSSL_ENCODER_R_MISSING_GET_PARAMS) {
        fprintf(stderr, "Encoder is missing get_params implementation\n");
    }
}
```
