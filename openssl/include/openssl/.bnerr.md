## Usage Example

```c
#include <openssl/bn.h>
#include <openssl/bnerr.h>
#include <openssl/err.h>

void check_bn_error(void) {
    unsigned long err = ERR_get_error();
    int reason = ERR_GET_REASON(err);

    switch (reason) {
        case BN_R_DIV_BY_ZERO:
            fprintf(stderr, "BN error: division by zero\n");
            break;
        case BN_R_NO_INVERSE:
            fprintf(stderr, "BN error: no modular inverse\n");
            break;
        case BN_R_INPUT_NOT_REDUCED:
            fprintf(stderr, "BN error: input not reduced\n");
            break;
        default:
            fprintf(stderr, "BN error: code %d\n", reason);
    }
}
```
