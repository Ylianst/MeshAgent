    unsigned long err = ERR_get_error();

    if (ERR_GET_LIB(err) == ERR_LIB_UI) {
        int reason = ERR_GET_REASON(err);

        if (reason == UI_R_NO_RESULT_BUFFER) {
            fprintf(stderr, "UI error: no result buffer allocated\n");
        } else if (reason == UI_R_RESULT_TOO_LARGE) {
            fprintf(stderr, "UI error: input exceeded buffer capacity\n");

    }
}
