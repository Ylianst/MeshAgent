## Usage Example

```c
#include <openssl/comp.h>
#include <openssl/err.h>
#include <openssl/comperr.h>

void handle_comp_error(void) {
    unsigned long err = ERR_get_error();
    int reason = ERR_GET_REASON(err);

    switch (reason) {
        case COMP_R_ZLIB_DEFLATE_ERROR:
            fprintf(stderr, "zlib deflate failed\n");
            break;
        case COMP_R_BROTLI_NOT_SUPPORTED:
            fprintf(stderr, "Brotli support not compiled in\n");
            break;
        case COMP_R_ZSTD_COMPRESS_ERROR:
            fprintf(stderr, "zstd compression failed\n");
            break;
        default:
            ERR_print_errors_fp(stderr);
    }
}
```
