    unsigned long err = ERR_get_error();
    int reason = ERR_GET_REASON(err);

    if (reason == CRMF_R_NULL_ARGUMENT) {
        fprintf(stderr, "CRMF error: null argument passed\n");
    } else if (reason == CRMF_R_UNSUPPORTED_ALGORITHM) {
        fprintf(stderr, "CRMF error: algorithm not supported\n");
    }
}
