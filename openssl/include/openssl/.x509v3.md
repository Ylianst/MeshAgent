## Usage Example

```c
#include <openssl/x509v3.h>

/* Build a context, then read a SAN extension from a certificate */
STACK_OF(GENERAL_NAME) *sans =
    X509_get_ext_d2i(cert, NID_subject_alt_name, NULL, NULL);

for (int i = 0; i < sk_GENERAL_NAME_num(sans); i++) {
    GENERAL_NAME *gn = sk_GENERAL_NAME_value(sans, i);
    if (gn->type == GEN_DNS) {
        /* gn->d.dNSName is an ASN1_IA5STRING */
        printf("DNS SAN: %s\n", gn->d.dNSName->data);
    } else if (gn->type == GEN_IPADD) {
        /* gn->d.iPAddress is a 4- or 16-byte ASN1_OCTET_STRING */
    }
}
GENERAL_NAMES_free(sans);

/* Register a custom extension method */
X509V3_EXT_METHOD my_method = {
    .ext_nid  = my_nid,
    .it       = ASN1_ITEM_ref(MY_EXT_TYPE),
    /* callbacks: i2s, s2i, i2r, r2i, etc. */
};
X509V3_EXT_add(&my_method);
```
