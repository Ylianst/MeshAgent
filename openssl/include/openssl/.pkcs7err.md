```c
#include <openssl/pkcs7.h>
#include <openssl/pkcs7err.h>
#include <openssl/err.h>

int verify_pkcs7(PKCS7 *p7, X509_STORE *store) {
    if (PKCS7_verify(p7, NULL, store, NULL, NULL, 0) != 1) {
        unsigned long err = ERR_get_error();

        /* Check for a specific PKCS7 reason code */
        if (ERR_GET_REASON(err) == PKCS7_R_CERTIFICATE_VERIFY_ERROR) {
            fprintf(stderr, "Certificate verification failed\n");
        } else if (ERR_GET_REASON(err) == PKCS7_R_SIGNATURE_FAILURE) {
            fprintf(stderr, "Signature validation failed\n");
        }
        return 0;
    }
    return 1;
}
```
